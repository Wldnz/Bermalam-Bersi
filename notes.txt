shortcut unutk menambahkan path

export PATH=$PATH:$(go env GOPATH)/bin

perintah untuk menjalakan golang migrate (goose)

goose -dir [nama_folder] create [nama_file] [format_file: sql/golang]

## kalo butuh sequel number
goose -s -dir [nama_folder] create [nama_file] [format_file: sql/golang]

untuk lebih cepet exec command
goose -dir migrations mysql "user:password@/dbname" [command]
